export type TaskRow = {
  id: string;
  title: string;
  dueOn: string | null;
  status: string;
  source: string;
  brandId: string | null;
  company: string | null;
};

export async function listTasks(ctx: Ctx, ownerId?: string | null): Promise<TaskRow[]> {
  const owner = scopeOwner(ctx, ownerId);
  const eq: Record<string, string> = { event_id: ctx.eventId };
  if (owner) eq.owner_id = owner;
  const rows = (await fetchAll<Record<string, unknown>>("tasks", "id, title, due_date, status, source, brand_id", eq)).filter(
    (r) => r.status === "todo" || r.status === "doing",
  );
  const ids = [...new Set(rows.map((r) => r.brand_id).filter((x): x is string => typeof x === "string"))];
  const names = new Map<string, string>();
  const supabase = getServiceClient();
  for (let i = 0; i < ids.length; i += 100) {
    const { data, error } = await supabase.from("brands").select("id, name").in("id", ids.slice(i, i + 100));
    if (error) throw new Error(`brands: ${error.message}`);
    for (const b of data ?? []) names.set(String(b.id), String(b.name));
  }
  return rows
    .map((r) => ({
      id: String(r.id),
      title: String(r.title),
      dueOn: (r.due_date as string | null) ?? null,
      status: String(r.status),
      source: String(r.source ?? "manual"),
      brandId: (r.brand_id as string | null) ?? null,
      company: typeof r.brand_id === "string" ? names.get(r.brand_id) ?? null : null,
    }))
    .sort((a, b) => (a.dueOn ?? "9999").localeCompare(b.dueOn ?? "9999") || a.title.localeCompare(b.title));
}

export async function createTask(ctx: Ctx, input: { title: string; dueOn?: string | null; brandId?: string | null }): Promise<TaskRow> {
  assertWritable(ctx);
  const memberId = requireMember(ctx);
  const title = input.title.trim();
  if (!title) throw new ValidationError("A task needs a title.");
  const holidays = await getHolidays(ctx.eventId);
  const dueOn = input.dueOn ?? addBusinessDays(ctx.todayIst, 1, holidays);
  const supabase = getServiceClient();
  const { data, error } = await supabase
    .from("tasks")
    .insert({
      event_id: ctx.eventId,
      title,
      owner_id: memberId,
      created_by: memberId,
      brand_id: input.brandId ?? null,
      due_date: dueOn,
      status: "todo",
      source: "manual",
    })
    .select("id, title, due_date, status, source, brand_id")
    .single();
  if (error) throw new Error(`tasks: ${error.message}`);
  return {
    id: String(data.id),
    title: String(data.title),
    dueOn: (data.due_date as string | null) ?? null,
    status: String(data.status),
    source: String(data.source),
    brandId: (data.brand_id as string | null) ?? null,
    company: null,
  };
}

export async function setTaskDone(ctx: Ctx, taskId: string, done: boolean): Promise<void> {
  assertWritable(ctx);
  const supabase = getServiceClient();
  const { data, error } = await supabase.from("tasks").select("id, owner_id, event_id").eq("id", taskId).maybeSingle();
  if (error) throw new Error(`tasks: ${error.message}`);
  if (!data || data.event_id !== ctx.eventId) throw new NotFoundError("Task not found.");
  if (ctx.role === "coordinator" && data.owner_id !== ctx.memberId) throw new ForbiddenError("This task belongs to someone else.");
  const { error: upErr } = await supabase
    .from("tasks")
    .update({ status: done ? "done" : "todo", completed_at: done ? new Date().toISOString() : null })
    .eq("id", taskId);
  if (upErr) throw new Error(`tasks: ${upErr.message}`);
}

import { getServiceClient } from "@/lib/db/client";

import type { Ctx, DealDeskRow } from "@/lib/types";

export const OPEN_DEAL_STAGES = ["prospecting", "proposal_sent", "negotiating"] as const;
export type OpenDealStage = (typeof OPEN_DEAL_STAGES)[number];

/** Coordinators see their own open deals; closers see every open deal. */
export async function listOpenDeals(ctx: Ctx): Promise<DealDeskRow[]> {
  return getDealDesk(ctx);
}

async function loadDeal(ctx: Ctx, dealId: string): Promise<{ id: string; brandId: string; stage: string }> {
  const { data, error } = await getServiceClient().from("deals").select("id, event_id, owner_id, brand_id, stage").eq("id", dealId).maybeSingle();
  if (error) throw new Error(`deals: ${error.message}`);
  if (!data || data.event_id !== ctx.eventId) throw new NotFoundError("Deal not found.");
  if (ctx.role === "coordinator" && !ctx.isCloser && data.owner_id !== ctx.memberId) throw new ForbiddenError("This deal belongs to another coordinator.");
  if (!(OPEN_DEAL_STAGES as readonly string[]).includes(String(data.stage))) throw new ValidationError("This deal is already closed.");
  return { id: String(data.id), brandId: String(data.brand_id), stage: String(data.stage) };
}

/** Closing a deal goes through the company status (Converted or Rejected), so only open stages are editable here. */
export async function updateDeal(ctx: Ctx, dealId: string, patch: { stage?: OpenDealStage; nextFollowUp?: string | null }): Promise<void> {
  assertWritable(ctx);
  const deal = await loadDeal(ctx, dealId);
  const supabase = getServiceClient();
  const now = new Date().toISOString();
  const update: Record<string, unknown> = {};
  if (patch.stage && patch.stage !== deal.stage) {
    update.stage = patch.stage;
    update.stage_changed_at = now;
  }
  if (patch.nextFollowUp !== undefined) update.next_follow_up = patch.nextFollowUp;
  if (Object.keys(update).length === 0) return;
  const { error } = await supabase.from("deals").update(update).eq("id", deal.id);
  if (error) throw new Error(`deals: ${error.message}`);
  const brandPatch: Record<string, unknown> = {};
  if (update.stage) brandPatch.last_touch_at = now;
  if (patch.nextFollowUp !== undefined) brandPatch.next_follow_up = patch.nextFollowUp;
  const { error: bErr } = await supabase.from("brands").update(brandPatch).eq("id", deal.brandId);
  if (bErr) throw new Error(`brands: ${bErr.message}`);
}

/** One nudge: logs a follow-up activity and resets the idle clock. */
export async function nudgeDeal(ctx: Ctx, dealId: string): Promise<void> {
  assertWritable(ctx);
  const deal = await loadDeal(ctx, dealId);
  const supabase = getServiceClient();
  const now = new Date().toISOString();
  const { error } = await supabase.from("activities").insert({
    event_id: ctx.eventId,
    brand_id: deal.brandId,
    owner_id: ctx.memberId,
    event_type: "follow_up",
    summary: "Nudge sent from the deal desk",
    occurred_at: now,
  });
  if (error) throw new Error(`activities: ${error.message}`);
  const { error: bErr } = await supabase.from("brands").update({ last_touch_at: now }).eq("id", deal.brandId);
  if (bErr) throw new Error(`brands: ${bErr.message}`);
}

import { getServiceClient } from "@/lib/db/client";
import { assertWritable } from "@/lib/db/guards";
import { ForbiddenError, NotFoundError, ValidationError } from "@/lib/db/errors";
import { addBusinessDays, addDays, toIstDate } from "@/lib/domain/bizdays";
import { validateTransition, type StatusChangeInput } from "@/lib/domain/crm-rules";
import type { StatusCode } from "@/lib/domain/status";
import type { Ctx } from "@/lib/types";
import { getHolidays } from "./holidays";

type BrandState = {
  id: string;
  name: string;
  owner_id: string | null;
  crm_status: StatusCode;
  call_attempts: number;
  first_outreach_at: string | null;
  first_response_at: string | null;
};

async function loadOwnedBrand(ctx: Ctx, brandId: string): Promise<BrandState> {
  const supabase = getServiceClient();
  const { data, error } = await supabase
    .from("brands")
    .select("id, name, event_id, owner_id, crm_status, call_attempts, first_outreach_at, first_response_at")
    .eq("id", brandId)
    .maybeSingle();
  if (error) throw new Error(`brands: ${error.message}`);
  if (!data || data.event_id !== ctx.eventId || !data.crm_status) throw new NotFoundError("Company not found.");
  if (ctx.role === "coordinator" && data.owner_id !== ctx.memberId) throw new ForbiddenError("This company belongs to another coordinator.");
  return {
    id: String(data.id),
    name: String(data.name),
    owner_id: (data.owner_id as string | null) ?? null,
    crm_status: data.crm_status as StatusCode,
    call_attempts: num(data.call_attempts),
    first_outreach_at: (data.first_outreach_at as string | null) ?? null,
    first_response_at: (data.first_response_at as string | null) ?? null,
  };
}

/** Turn a database guard message into a friendly validation error. */
function friendly(message: string): Error {
  if (message.includes("Didn") || message.includes("Ghosting")) return new ValidationError(message);
  return new Error(message);
}

const OUTREACH_STATUSES: StatusCode[] = ["slabs_sent", "didnt_pick", "contacted", "ghosting", "prospective", "converted", "rejected"];
const RESPONSE_STATUSES: StatusCode[] = ["contacted", "prospective", "converted", "rejected"];

async function logStatusActivity(ctx: Ctx, brand: BrandState, note: string | null, to: StatusCode): Promise<void> {
  const supabase = getServiceClient();
  await supabase.from("activities").insert({
    event_id: ctx.eventId,
    brand_id: brand.id,
    owner_id: brand.owner_id,
    event_type: "status_change",
    outcome: to,
    summary: note,
    occurred_at: new Date().toISOString(),
  });
}

async function createPaymentRows(ctx: Ctx, brandId: string): Promise<{ dealId: string | null; created: number }> {
  const supabase = getServiceClient();
  const { data: deals, error } = await supabase
    .from("deals")
    .select("id, signed_cash_inr, stage_changed_at")
    .eq("brand_id", brandId)
    .eq("stage", "partner_confirmed")
    .order("stage_changed_at", { ascending: false })
    .limit(1);
  if (error) throw new Error(`deals: ${error.message}`);
  const deal = deals?.[0];
  if (!deal) return { dealId: null, created: 0 };
  const dealId = String(deal.id);
  const { count, error: cErr } = await supabase.from("payments").select("id", { count: "exact", head: true }).eq("deal_id", dealId);
  if (cErr) throw new Error(`payments: ${cErr.message}`);
  if ((count ?? 0) > 0) return { dealId, created: 0 };
  const cash = num(deal.signed_cash_inr);
  if (cash <= 0) return { dealId, created: 0 };
  const plan = ctx.event.model_assumptions.plan;
  const raw = plan && typeof plan === "object" ? (plan as Record<string, unknown>).prefestShare : undefined;
  const share = typeof raw === "number" && raw > 0 && raw < 1 ? raw : 0.75;
  const pre = Math.round(cash * share);
  const start = ctx.event.starts_at ? toIstDate(ctx.event.starts_at) : ctx.todayIst;
  const end = ctx.event.ends_at ? toIstDate(ctx.event.ends_at) : start;
  const rows = [
    { deal_id: dealId, amount_inr: pre, due_date: addDays(start, -1), phase: "pre_fest", note: "Pre-fest instalment (assumed 75%)" },
    { deal_id: dealId, amount_inr: cash - pre, due_date: addDays(end, 15), phase: "post_fest", note: "Post-fest instalment (assumed 25%)" },
  ];
  const { error: iErr } = await supabase.from("payments").insert(rows);
  if (iErr) throw new Error(`payments: ${iErr.message}`);
  return { dealId, created: rows.length };
}

export async function changeCompanyStatus(ctx: Ctx, input: StatusChangeInput): Promise<{ dealId: string | null; paymentsCreated: number }> {
  assertWritable(ctx);
  const brand = await loadOwnedBrand(ctx, input.brandId);
  const problem = validateTransition(brand.crm_status, input);
  if (problem) throw new ValidationError(problem);

  const now = new Date().toISOString();
  const patch: Record<string, unknown> = { crm_status: input.to, last_touch_at: now };
  if (OUTREACH_STATUSES.includes(input.to) && !brand.first_outreach_at) patch.first_outreach_at = now;
  if (RESPONSE_STATUSES.includes(input.to) && !brand.first_response_at) patch.first_response_at = now;
  if (input.note) patch.notes = input.note;

  if (input.to === "prospective") {
    patch.approx_amount_inr = input.approxAmountInr;
    const holidays = await getHolidays(ctx.eventId);
    patch.next_follow_up = input.nextFollowUp ?? addBusinessDays(ctx.todayIst, 3, holidays);
  } else if (input.nextFollowUp !== undefined) {
    patch.next_follow_up = input.nextFollowUp;
  }
  if (input.to === "converted") {
    patch.final_amount_inr = input.finalAmountInr;
    patch.final_in_kind_inr = input.inKindInr ?? 0;
    patch.next_follow_up = null;
  }
  if (input.to === "rejected") {
    patch.rejection_reason = (input.rejectionReason ?? "").trim();
    patch.next_follow_up = null;
  }

  const supabase = getServiceClient();
  const { error } = await supabase.from("brands").update(patch).eq("id", brand.id);
  if (error) throw friendly(error.message);
  await logStatusActivity(ctx, brand, input.note ?? null, input.to);

  if (input.to === "converted") {
    const res = await createPaymentRows(ctx, brand.id);
    return { dealId: res.dealId, paymentsCreated: res.created };
  }
  return { dealId: null, paymentsCreated: 0 };
}

export type ActivityKind = "call" | "call_no_answer" | "note";

/** Quick actions. A call with no answer adds an attempt but never changes the status. */
export async function logActivity(
  ctx: Ctx,
  input: { brandId: string; kind: ActivityKind; note?: string | null; nextFollowUp?: string | null },
): Promise<{ attempts: number }> {
  assertWritable(ctx);
  const brand = await loadOwnedBrand(ctx, input.brandId);
  const supabase = getServiceClient();
  const now = new Date().toISOString();
  const attempts = input.kind === "call_no_answer" ? brand.call_attempts + 1 : brand.call_attempts;
  const { error: aErr } = await supabase.from("activities").insert({
    event_id: ctx.eventId,
    brand_id: brand.id,
    owner_id: brand.owner_id,
    event_type: input.kind === "note" ? "note" : "call_attempt",
    outcome: input.kind === "call" ? "answered" : input.kind === "call_no_answer" ? "no_answer" : null,
    summary: input.note ?? null,
    occurred_at: now,
  });
  if (aErr) throw new Error(`activities: ${aErr.message}`);
  const patch: Record<string, unknown> = { last_touch_at: now, call_attempts: attempts };
  if (input.nextFollowUp !== undefined) patch.next_follow_up = input.nextFollowUp;
  const { error } = await supabase.from("brands").update(patch).eq("id", brand.id);
  if (error) throw new Error(`brands: ${error.message}`);
  return { attempts };
}

export async function setFollowUp(ctx: Ctx, brandId: string, date: string | null): Promise<void> {
  assertWritable(ctx);
  const brand = await loadOwnedBrand(ctx, brandId);
  const { error } = await getServiceClient().from("brands").update({ next_follow_up: date }).eq("id", brand.id);
  if (error) throw new Error(`brands: ${error.message}`);
}
