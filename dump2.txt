"use client";

import { useRouter } from "next/navigation";
import { useTransition } from "react";

export function MemberPicker(props: { members: { id: string; name: string }[]; memberId: string | null }) {
  const router = useRouter();
  const [pending, startTransition] = useTransition();

  async function choose(id: string) {
    const res = await fetch("/api/session", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ memberId: id }),
    });
    if (res.ok) startTransition(() => router.refresh());
  }

  return (
    <label className="flex items-center gap-2 text-sm text-muted">
      <span>Coordinator</span>
      <select
        value={props.memberId ?? ""}
        disabled={pending}
        onChange={(event) => void choose(event.target.value)}
        className="h-9 rounded-lg border border-border bg-card px-2 text-foreground focus-visible:outline focus-visible:outline-2 focus-visible:outline-accent"
      >
        {props.members.map((member) => (
          <option key={member.id} value={member.id}>
            {member.name}
          </option>
        ))}
      </select>
    </label>
  );
}

=== app\api\session\route.ts ===
import { cookies } from "next/headers";
import { z } from "zod";
import { EVENT_COOKIE, MEMBER_COOKIE, ROLE_COOKIE } from "@/lib/session";

const bodySchema = z.object({
  role: z.enum(["head", "coordinator"]).optional(),
  eventId: z.enum(["saarang-2027", "campusfest-2025-work", "campusfest-2025"]).optional(),
  memberId: z.string().min(1).nullable().optional(),
});

const OPTIONS = { path: "/", maxAge: 60 * 60 * 24 * 30, sameSite: "lax" as const, httpOnly: true };

export async function POST(request: Request) {
  const parsed = bodySchema.safeParse(await request.json().catch(() => null));
  if (!parsed.success) return Response.json({ ok: false, error: "Invalid body" }, { status: 400 });
  const { role, eventId, memberId } = parsed.data;
  const store = await cookies();
  if (role) store.set(ROLE_COOKIE, role, OPTIONS);
  if (eventId) store.set(EVENT_COOKIE, eventId, OPTIONS);
  if (memberId === null) store.delete(MEMBER_COOKIE);
  else if (memberId) store.set(MEMBER_COOKIE, memberId, OPTIONS);
  return Response.json({ ok: true });
}

import * as React from "react";
import { Slot } from "@radix-ui/react-slot";
import { cva, type VariantProps } from "class-variance-authority";
import { cn } from "@/lib/utils";

const buttonVariants = cva(
  "inline-flex items-center justify-center gap-2 rounded-[16px] text-sm font-medium transition-colors disabled:pointer-events-none disabled:opacity-50",
  {
    variants: {
      variant: {
        default: "bg-accent text-background hover:bg-accent/90",
        outline: "border border-border bg-card hover:bg-border/40",
        ghost: "hover:bg-border/30",
        danger: "bg-danger text-background",
      },
      size: {
        default: "h-11 px-4",
        sm: "h-9 px-3",
      },
    },
    defaultVariants: {
      variant: "default",
      size: "default",
    },
  },
);

export function Button({
  className,
  variant,
  size,
  asChild = false,
  ...props
}: React.ComponentProps<"button"> &
  VariantProps<typeof buttonVariants> & { asChild?: boolean }) {
  const Comp = asChild ? Slot : "button";
  return <Comp className={cn(buttonVariants({ variant, size, className }))} {...props} />;
}
