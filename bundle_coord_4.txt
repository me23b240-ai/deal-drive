import { cookies } from "next/headers";
import { z } from "zod";
import { EVENT_COOKIE, MEMBER_COOKIE, ROLE_COOKIE } from "@/lib/session";

const bodySchema = z.object({
  role: z.enum(["head", "coordinator"]).optional(),
  eventId: z.enum(["saarang-2027", "campusfest-2025-work", "campusfest-2025"]).optional(),
  memberId: z.string().min(1).nullable().optional(),
});

const OPTIONS = { path: "/", maxAge: 60 * 60 * 24 * 30, sameSite: "lax" as const, httpOnly: true };

export async function POST(request: Request) {
  const parsed = bodySchema.safeParse(await request.json().catch(() => null));
  if (!parsed.success) return Response.json({ ok: false, error: "Invalid body" }, { status: 400 });
  const { role, eventId, memberId } = parsed.data;
  const store = await cookies();
  if (role) store.set(ROLE_COOKIE, role, OPTIONS);
  if (eventId) store.set(EVENT_COOKIE, eventId, OPTIONS);
  if (memberId === null) store.delete(MEMBER_COOKIE);
  else if (memberId) store.set(MEMBER_COOKIE, memberId, OPTIONS);
  return Response.json({ ok: true });
}

import { useRouter } from "next/navigation";
import { useTransition, type KeyboardEvent } from "react";
import type { DemoRole } from "@/lib/session";

export function RoleToggle(props: { role: DemoRole }) {
  const router = useRouter();
  const [pending, startTransition] = useTransition();
  const isHead = props.role === "head";

  async function choose(next: DemoRole) {
    if (next === props.role || pending) return;
    const res = await fetch("/api/session", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ role: next }),
    });
    if (!res.ok) return;
    startTransition(() => {
      router.push(next === "head" ? "/head" : "/coordinator");
      router.refresh();
    });
  }

  function onKeyDown(event: KeyboardEvent<HTMLDivElement>) {
    if (event.key === "ArrowLeft") {
      event.preventDefault();
      void choose("head");
    } else if (event.key === "ArrowRight") {
      event.preventDefault();
      void choose("coordinator");
    }
  }

  const base =
    "relative z-10 flex h-full flex-1 items-center justify-center rounded-full text-sm focus-visible:outline focus-visible:outline-2 focus-visible:outline-accent";

  return (
    <div
      role="radiogroup"
      aria-label="Role"
      onKeyDown={onKeyDown}
      className="relative flex h-11 min-h-11 w-[220px] items-center rounded-full border-2 border-accent/40 bg-card p-1"
    >
      <span
        aria-hidden
        className="absolute top-1 bottom-1 w-[calc(50%-4px)] rounded-full bg-accent/20 transition-transform duration-200 motion-reduce:transition-none"
        style={{ transform: isHead ? "translateX(0)" : "translateX(100%)" }}
      />
      <button
        type="button"
        role="radio"
        aria-checked={isHead}
        tabIndex={isHead ? 0 : -1}
        onClick={() => void choose("head")}
        className={`${base} ${isHead ? "text-foreground" : "text-muted"}`}
      >
        Head
      </button>
      <button
        type="button"
        role="radio"
        aria-checked={!isHead}
        tabIndex={!isHead ? 0 : -1}
        onClick={() => void choose("coordinator")}
        className={`${base} ${!isHead ? "text-foreground" : "text-muted"}`}
      >
        Coordinator
      </button>
    </div>
  );
}

export class HttpError extends Error {
  readonly status: number;
  constructor(status: number, message: string) {
    super(message);
    this.name = "HttpError";
    this.status = status;
  }
}

export class ForbiddenError extends HttpError {
  constructor(message = "Not allowed") {
    super(403, message);
    this.name = "ForbiddenError";
  }
}

export class NotFoundError extends HttpError {
  constructor(message = "Not found") {
    super(404, message);
    this.name = "NotFoundError";
  }
}

export class ValidationError extends HttpError {
  constructor(message = "Invalid input") {
    super(400, message);
    this.name = "ValidationError";
  }
}

export function errorStatus(error: unknown): number {
  return error instanceof HttpError ? error.status : 500;
}

export function errorMessage(error: unknown): string {
  if (error instanceof Error) return error.message;
  if (error && typeof error === "object" && "message" in error) {
    const message = (error as { message: unknown }).message;
    if (typeof message === "string") return message;
  }
  return "Unknown error";
}

import { cache } from "react";
import { getSession, DEFAULT_EVENT_ID, type DemoRole } from "@/lib/session";
import { toIstDate, todayIST } from "@/lib/domain/bizdays";
import { fallbackEvent, getEventInfo } from "@/lib/db/queries/events";
import { listTeam } from "@/lib/db/queries/team";
import type { Ctx } from "@/lib/types";

/** Pages pass their route role ("head" or "coordinator") so the URL decides the role. */
export const getCtx = cache(async (routeRole?: DemoRole): Promise<Ctx> => {
  const session = await getSession();
  const role = routeRole ?? session.role;

  let event = null;
  try {
    event = (await getEventInfo(session.eventId)) ?? (await getEventInfo(DEFAULT_EVENT_ID));
  } catch {
    event = null;
  }
  const info = event ?? fallbackEvent(session.eventId);

  let memberId = session.memberId;
  let isCloser = false;
  if (role === "coordinator") {
    const team = await listTeam(info.id).catch(() => []);
    const coordinators = team.filter((m) => m.role === "member");
    const mine = coordinators.find((m) => m.id === memberId) ?? coordinators[0];
    memberId = mine?.id ?? null;
    isCloser = mine?.is_closer ?? false;
  }

  return {
    eventId: info.id,
    event: info,
    role,
    memberId,
    isCloser,
    todayIst: info.as_of ? toIstDate(info.as_of) : todayIST(),
    isLocked: info.is_seed,
  };
});

/** Default next follow-up: today plus the status threshold in business days (null if no rule). */
export async function getDefaultFollowUp(ctx: Ctx, status: StatusCode): Promise<string | null> {
  const [rules, holidays] = await Promise.all([getStaleRules(ctx), getHolidays(ctx.eventId)]);
  const rule = rules.find((r) => r.state_key === status);
  return rule ? addBusinessDays(ctx.todayIst, rule.days_threshold, holidays) : null;
}

async function viewRows<T>(view: string, eventId: string, ownerId: string | null): Promise<T[]> {
  const supabase = getServiceClient();
  let query = supabase.from(view).select("*").eq("event_id", eventId);
  if (ownerId) query = query.eq("owner_id", ownerId);
  const { data, error } = await query;
  if (error) throw new Error(`${view}: ${error.message}`);

}

export async function getWeeklyStrip(ctx: Ctx, memberId?: string | null): Promise<WeeklyStrip> {
  const owner = scopeOwner(ctx, memberId ?? ctx.memberId);
  if (!owner) throw new ValidationError("A coordinator must be selected.");
  const weekStart = weekStartIST(ctx.todayIst);
  const [holidays, tgt, progress] = await Promise.all([
    getHolidays(ctx.eventId),
    getWeeklyTarget(ctx, owner, weekStart),
    getWeeklyProgress(ctx, owner),
  ]);
  const row = progress.find((r) => r.week_start === weekStart);
  const outreached = row?.outreached ?? 0;
  const responded = row?.responded ?? 0;
  const left = businessDaysLeftInWeek(ctx.todayIst, holidays);
  const remaining = Math.max(0, tgt.target - outreached);
  return {
    weekStart,
    weekEnd: weekEndIST(weekStart),
    target: tgt.target,
    label: tgt.label,
    source: tgt.source,
    outreached,
    responded,
    responseRate: outreached > 0 ? responded / outreached : null,
    businessDaysLeft: left,
    remaining,
    requiredPerDay: left > 0 ? remaining / left : null,
  };
}

export type Holiday = { day: string };

const DAY_MS = 86_400_000;
const istFormat = new Intl.DateTimeFormat("en-CA", {
  timeZone: "Asia/Kolkata",
  year: "numeric",
  month: "2-digit",
  day: "2-digit",
});

function parseIso(iso: string): number {
  const [y, m, d] = iso.split("-").map(Number);
  return Date.UTC(y, m - 1, d);
}

function formatMs(ms: number): string {
  const d = new Date(ms);
  const y = d.getUTCFullYear();
  const m = String(d.getUTCMonth() + 1).padStart(2, "0");
  const day = String(d.getUTCDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

function holidaySet(holidays: Holiday[]): Set<string> {
  return new Set(holidays.map((h) => h.day));
}

export function toIstDate(value: string | Date): string {
  return istFormat.format(typeof value === "string" ? new Date(value) : value);
}

export function todayIST(now: Date = new Date()): string {
  return istFormat.format(now);
}

export function addDays(iso: string, days: number): string {
  return formatMs(parseIso(iso) + days * DAY_MS);
}

export function daysBetween(start: string, end: string): number {
  return Math.round((parseIso(end) - parseIso(start)) / DAY_MS);
}

export function isoDow(iso: string): number {
  const dow = new Date(parseIso(iso)).getUTCDay();
  return dow === 0 ? 7 : dow;
}

export function isBusinessDay(iso: string, holidays: Holiday[] = []): boolean {
  return isoDow(iso) <= 5 && !holidaySet(holidays).has(iso);
}

/** Business days AFTER start, up to and including end (matches SQL business_days_between). */
export function businessDaysBetween(start: string, end: string, holidays: Holiday[] = []): number {
  if (end <= start) return 0;
  const hs = holidaySet(holidays);
  const last = parseIso(end);
  let count = 0;
  for (let ms = parseIso(start) + DAY_MS; ms <= last; ms += DAY_MS) {
    const dow = new Date(ms).getUTCDay();
    if (dow !== 0 && dow !== 6 && !hs.has(formatMs(ms))) count += 1;
  }
  return count;
}

export function addBusinessDays(start: string, days: number, holidays: Holiday[] = []): string {
  const hs = holidaySet(holidays);
  let ms = parseIso(start);
  let remaining = Math.max(0, Math.floor(days));
  while (remaining > 0) {
    ms += DAY_MS;
    const dow = new Date(ms).getUTCDay();
    if (dow !== 0 && dow !== 6 && !hs.has(formatMs(ms))) remaining -= 1;
  }
  return formatMs(ms);
}

export function weekStartIST(iso: string): string {
  return addDays(iso, -(isoDow(iso) - 1));
}

export function weekEndIST(iso: string): string {
  return addDays(weekStartIST(iso), 6);
}

/** Business days left in the Mon-Sun week, counting today if it is a business day. */
export function businessDaysLeftInWeek(iso: string, holidays: Holiday[] = []): number {
  return businessDaysBetween(iso, weekEndIST(iso), holidays) + (isBusinessDay(iso, holidays) ? 1 : 0);
}

const RUPEE = "\u20B9";

function groupIndian(amount: number): string {
  const digits = Math.abs(Math.round(amount)).toString();
  if (digits.length <= 3) return digits;
  const last3 = digits.slice(-3);
  const rest = digits.slice(0, -3).replace(/\B(?=(\d{2})+(?!\d))/g, ",");
  return `${rest},${last3}`;
}

export function formatInr(amount: number | null | undefined): string {
  if (amount === null || amount === undefined || Number.isNaN(amount)) return "-";
  const sign = Math.round(amount) < 0 ? "-" : "";
  return `${sign}${RUPEE}${groupIndian(amount)}`;
}

export function formatLakh(amount: number | null | undefined): string {
  if (amount === null || amount === undefined || Number.isNaN(amount)) return "-";
  const sign = amount < 0 ? "-" : "";
  return `${sign}${RUPEE}${(Math.abs(amount) / 100_000).toFixed(1)}L`;
}

export function parseInr(input: string): number | null {
  const cleaned = input.replace(/[\u20B9,\s]/g, "").toLowerCase();
  if (!cleaned) return null;
  const lakh = cleaned.endsWith("l");
  const value = Number(lakh ? cleaned.slice(0, -1) : cleaned);
  if (!Number.isFinite(value)) return null;
  return Math.round(lakh ? value * 100_000 : value);
}

function isObj(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function merge<T>(defaults: T, raw: unknown): T {
  if (isObj(defaults)) {
    const src = isObj(raw) ? raw : {};
    const out: Record<string, unknown> = {};
    for (const key of Object.keys(defaults)) out[key] = merge(defaults[key], src[key]);
    return out as T;
  }
  if (raw === undefined) return defaults;
  if (defaults === null) return (typeof raw === "string" || raw === null ? raw : defaults) as T;
  if (typeof defaults === "number") {
    return (typeof raw === "number" && Number.isFinite(raw) ? raw : defaults) as T;
  }
  return (typeof raw === typeof defaults ? raw : defaults) as T;
}

export function resolveAssumptions(raw: unknown): ModelAssumptions {
  const merged = merge(DEFAULT_ASSUMPTIONS, raw);
  if (merged.inkind_policy_scope !== "all" && merged.inkind_policy_scope !== "new_only") {
    merged.inkind_policy_scope = "all";
  }

-- Business days between (exclusive start, inclusive end): Mon-Fri minus holidays.
create or replace function business_days_between(p_start date, p_end date, p_event_id text)
returns integer
language sql
stable
as $$
  select count(*)::integer
  from generate_series((p_start + 1), p_end, interval '1 day') as g(d)
  where extract(isodow from g.d) between 1 and 5
    and not exists (
      select 1 from holidays h
      where h.event_id = p_event_id and h.on_date = g.d::date
    );
$$;

create or replace function is_business_day(p_date date, p_event_id text)
returns boolean
language sql
stable
as $$
  select extract(isodow from p_date) between 1 and 5
    and not exists (
      select 1 from holidays h
      where h.event_id = p_event_id and h.on_date = p_date
    );
$$;

create or replace function brands_status_before()
returns trigger
language plpgsql
as $$
begin
  if tg_op = 'UPDATE' and new.crm_status is distinct from old.crm_status then
    if new.crm_status = 'didnt_pick' and old.crm_status not in ('to_contact', 'send_mail', 'didnt_pick') then
      raise exception using
        message = 'Didn''t Pick is only valid from To Contact, Send Mail, or Didn''t Pick',
        errcode = 'P0001';
    end if;
    if new.crm_status = 'ghosting' and old.crm_status not in ('contacted', 'prospective', 'ghosting') then
      raise exception using
        message = 'Ghosting is only valid from Contacted, Prospective, or Ghosting',
        errcode = 'P0001';
    end if;
    if new.crm_status in ('contacted', 'prospective', 'converted', 'rejected')
       and new.first_response_at is null then
      new.first_response_at := timezone('Asia/Kolkata', now());
    end if;
  end if;
  new.updated_at := now();
  return new;
end;
$$;

drop trigger if exists brands_status_before_trg on brands;
create trigger brands_status_before_trg
before update on brands
for each row execute function brands_status_before();

create or replace function brands_status_after()
returns trigger
language plpgsql
as $$
begin
  if new.crm_status is distinct from old.crm_status then
    insert into status_events (event_id, brand_id, from_status, to_status, member_id)
    values (new.event_id, new.id, old.crm_status, new.crm_status, new.owner_member_id);

    if new.crm_status = 'prospective' then
      insert into deals (event_id, brand_id, stage, approx_amount_inr, owner_member_id, next_follow_up)
      select new.event_id, new.id, 'prospecting', new.approx_amount_inr, new.owner_member_id, new.next_follow_up
      where not exists (
        select 1 from deals d
        where d.brand_id = new.id and d.stage in ('prospecting', 'proposal_sent', 'negotiating')
      );
    end if;

    if new.crm_status = 'converted' then
      update deals
      set
        stage = 'partner_confirmed',
        signed_cash_inr = coalesce(new.final_amount_inr, 0) - coalesce(new.final_in_kind_inr, 0),
        signed_in_kind_inr = coalesce(new.final_in_kind_inr, 0),
        confirmed_at = coalesce(confirmed_at, timezone('Asia/Kolkata', now()))
      where brand_id = new.id
        and stage in ('prospecting', 'proposal_sent', 'negotiating', 'partner_confirmed');

      if not found then
        insert into deals (
          event_id, brand_id, stage, approx_amount_inr,
          signed_cash_inr, signed_in_kind_inr, owner_member_id, confirmed_at
        ) values (
          new.event_id, new.id, 'partner_confirmed', new.approx_amount_inr,
          coalesce(new.final_amount_inr, 0) - coalesce(new.final_in_kind_inr, 0),
          coalesce(new.final_in_kind_inr, 0),
          new.owner_member_id,
          timezone('Asia/Kolkata', now())
        );
      end if;
    end if;

    if new.crm_status = 'rejected' then
      update deals
      set stage = 'lost'
      where brand_id = new.id
        and stage in ('prospecting', 'proposal_sent', 'negotiating');
    end if;
  end if;
  return new;
end;
$$;

drop trigger if exists brands_status_after_trg on brands;
create trigger brands_status_after_trg
after update on brands
for each row execute function brands_status_after();

-- Deal Drive 003: live views. Never store needs-update; compute it here.

create or replace view v_needs_update as
select
  b.*,
  business_days_between(coalesce(b.last_touch_at::date, b.created_at::date), (timezone('Asia/Kolkata', now()))::date, b.event_id) as idle_bdays,
  sr.business_days as stale_threshold,
  case
    when b.next_follow_up is not null and b.next_follow_up <= (timezone('Asia/Kolkata', now()))::date then 'follow-up due'
    else 'stale'
  end as flag_reason
from brands b
left join stale_rules sr
  on sr.event_id = b.event_id and sr.status_key = b.crm_status
where
  b.crm_status not in ('converted', 'rejected', 'left_company', 'wrong_number')
  and (
    (b.next_follow_up is not null and b.next_follow_up <= (timezone('Asia/Kolkata', now()))::date)
    or (
      b.next_follow_up is null
      and sr.business_days is not null
      and business_days_between(
        coalesce(b.last_touch_at::date, b.created_at::date),
        (timezone('Asia/Kolkata', now()))::date,
        b.event_id
      ) >= sr.business_days
    )
  );
