/** Pages pass their route role ("head" or "coordinator") so the URL decides the role. */
export const getCtx = cache(async (routeRole?: DemoRole): Promise<Ctx> => {
  const session = await getSession();
  const role = routeRole ?? session.role;

  let event = null;
  try {
    event = (await getEventInfo(session.eventId)) ?? (await getEventInfo(DEFAULT_EVENT_ID));
  } catch {
    event = null;
  }
  const info = event ?? fallbackEvent(session.eventId);

  let memberId = session.memberId;
  let isCloser = false;
  if (role === "coordinator") {
    const team = await listTeam(info.id).catch(() => []);
    const coordinators = team.filter((m) => m.role === "member");
    const mine = coordinators.find((m) => m.id === memberId) ?? coordinators[0];
    memberId = mine?.id ?? null;
    isCloser = mine?.is_closer ?? false;
  }

  return {
    eventId: info.id,
    event: info,
    role,
    memberId,
    isCloser,
    todayIst: info.as_of ? toIstDate(info.as_of) : todayIST(),
    isLocked: info.is_seed,
  };

/** A prospect with no touch for more than this many business days is stale. */
export const STALE_PROSPECT_BDAYS = 3;

/** Everything buildDecisions needs, read from the database. No payment rule: payments are 75/25 by phase, not due dates. */
export async function getDecisionSnapshot(ctx: Ctx): Promise<DecisionSnapshot> {
  requireHead(ctx);
  const [team, holidays, brands, week, gateReport, initiatives] = await Promise.all([
    listTeam(ctx.eventId),
    getHolidays(ctx.eventId),
    getBrandRows(ctx),
    getTeamWeekSummary(ctx),
    getGateReport(ctx),
    getInitiativeStatuses(ctx),
  ]);
  const nameOf = new Map(team.map((m) => [m.id, m.name]));
  const weekStart = weekStartIST(ctx.todayIst);
  const inWeek = businessDaysBetween(addDays(weekStart, -1), weekEndIST(weekStart), holidays);
  const left = businessDaysLeftInWeek(ctx.todayIst, holidays);

  const coordinators = coordinatorsOf(team).map((m) => {
    const w = week.members.find((x) => x.memberId === m.id);
    return coordinatorStat(
      { id: m.id, name: m.name },
      brands,
      { outreached: w?.outreached ?? 0, target: w?.target ?? ctx.event.weekly_target_default },
      ctx.todayIst,
      holidays,
      { inWeek, left },
    );
  });

  return {
    initiatives: initiatives.map((i) => ({ code: i.code, name: i.name, badge: i.badge, reason: i.reason })),
    coordinators,
    staleProspects: staleProspects(brands, STALE_PROSPECT_BDAYS, ctx.todayIst, holidays, nameOf),
    overduePayments: [],
    ownershipConflicts: ownershipConflicts(brands, nameOf),
    gates: gateReport.statuses,
  };
}

/** The five bets with their live badge. Rules come from campaigns.scale_rule / change_rule / stop_rule. */
export async function getInitiativeStatuses(ctx: Ctx): Promise<InitiativeStatus[]> {
  requireHead(ctx);
  const [campaigns, funnel, desk, signed, brands, drafts, events] = await Promise.all([
    fetchAll<Record<string, unknown>>("campaigns", "id, source_id, name, scale_rule, change_rule, stop_rule", {
      event_id: ctx.eventId,
      kind: "initiative",
    }),
    getInitiativeFunnel(ctx),
    getDealDesk(ctx),
    getSignedDeals(ctx),
    getBrandRows(ctx),
    fetchAll<Record<string, unknown>>("email_drafts", "brand_id, personalised, status", { event_id: ctx.eventId }),
    fetchAll<Record<string, unknown>>("status_events", "brand_id, to_status", { event_id: ctx.eventId }),
  ]);

  const today = dateNum(ctx.todayIst);
  const deskRows = desk.map((r) => ({
    company: r.company,
    stage: String(r.stage),
    proposed_inr: num(r.proposed_inr),
    idle_bdays: num(r.idle_bdays),
    nudges: num(r.nudges),
  }));
  const draftRows = drafts.map((r) => ({ brand_id: String(r.brand_id), personalised: Boolean(r.personalised), status: String(r.status) }));
  const eventRows = events.map((r) => ({ brand_id: String(r.brand_id), to_status: String(r.to_status) }));

  const out: InitiativeStatus[] = [];
  for (const c of campaigns) {
    const campaignId = String(c.id);
    const code = String(c.source_id);
    let set: MetricSet = { metrics: { today }, counts: {} };
    let experiment: Experiment | null = null;

    if (code === "bet-1") {
      set = dealDeskMetrics(deskRows, today);
    } else if (code === "bet-2") {
      set = cashTermsMetrics(signed, PLAN_START, today);
    } else {
      set = laneMetrics(funnel.find((f) => f.initiative_id === campaignId), today);
      if (code === "bet-5") {
        const ids = new Set(brands.filter((b) => b.initiative_id === campaignId).map((b) => b.id));
        experiment = experimentArms(ids, draftRows, eventRows);
        set = {
          metrics: { ...set.metrics, personalised_vs_template_ratio: experiment.ratio },
          counts: {
            ...set.counts,
            personalised_vs_template_ratio:
              experiment.ratio === undefined
                ? `personalised ${experiment.personalised.n} sent vs template ${experiment.template.n} sent`
                : `personalised ${experiment.personalised.reached} of ${experiment.personalised.n} vs template ${experiment.template.reached} of ${experiment.template.n} reached proposal`,
          },
        };
      }
    }

    const rules: InitiativeRules = { scale: asRule(c.scale_rule), change: asRule(c.change_rule), stop: asRule(c.stop_rule) };
    const evaluation = evaluateInitiative(code as CodeArg, rules, set.metrics, set.counts);
    out.push({
      campaignId,
      code,
      name: String(c.name),
      badge: evaluation.badge,
      reason: evaluation.reason,
      evaluation,
      metrics: set.metrics,
      experiment,
    });
  }
  return out.sort((a, b) => a.code.localeCompare(b.code));
}

import { getServiceClient } from "../client";

/** Reads every row of a table or view (1000 per page), filtered by equality. */
export async function fetchAll<T>(table: string, select: string, eq: Record<string, string> = {}): Promise<T[]> {
  const supabase = getServiceClient();
  const out: T[] = [];
  for (let from = 0; ; from += 1000) {
    let query = supabase.from(table).select(select).range(from, from + 999);
    for (const [column, value] of Object.entries(eq)) query = query.eq(column, value);
    const { data, error } = await query;
    if (error) throw new Error(`${table}: ${error.message}`);
    const rows = (data ?? []) as unknown as T[];
    out.push(...rows);
    if (rows.length < 1000) break;
  }
  return out;
}

async function viewRows<T>(view: string, eventId: string, ownerId: string | null): Promise<T[]> {
  const supabase = getServiceClient();
  let query = supabase.from(view).select("*").eq("event_id", eventId);
  if (ownerId) query = query.eq("owner_id", ownerId);
  const { data, error } = await query;
  if (error) throw new Error(`${view}: ${error.message}`);
  return (data ?? []) as T[];
}
