# Deal Drive architecture

Sponsorship command center for Saarang 2027 (simulated data) plus a CampusFest historical dataset.

## Layout

- `app/` App Router pages and route handlers. All mutating and reading UI goes through this layer.
- `app/api/*` HTTP API. Every handler validates input with zod, then calls `lib/db/queries` only.
- `components/` shell, coordinator, head, chat, and shadcn `ui/` primitives.
- `lib/db/queries` is the single data-access layer. Pages, APIs, and chatbot tools must not run ad-hoc SQL.
- `lib/domain` status rules, business days, money, channels, initiatives, rules engine, decisions.
- `lib/model` projection, attribution, and gates (pure functions, unit tested).
- `lib/ai` Gemini client, tools, draft guards, rate limit.
- `lib/seed` CampusFest xlsx loader and Saarang simulated seed.
- `scripts/` seed, verify, and chat golden tests (tsx, PowerShell-safe).
- `supabase/migrations/` `001_core.sql`, `002_events_and_facts.sql`, `003_views.sql`. Apply in the Supabase SQL editor.
- `tests/` vitest unit tests.
- `data/private/` gitignored. All Saarang CRM rows are fictional.

## Data flow

1. Demo session is cookies: role `head` | `coordinator`, member id, event id.
2. The server uses `SUPABASE_SERVICE_ROLE_KEY` and enforces scoping in queries. RLS is on with no policies (service role bypass). This is a demo limitation, not production auth.
3. Seeded `is_seed` events (`campusfest-2025`) reject mutations with 403.
4. Mutations call query helpers, then `revalidatePath` so Head and Coordinator see the same numbers.
5. Chatbot tools call the same query layer; write tools insert `pending_actions` and only execute on confirm.

## Time and money

- Timezone: Asia/Kolkata.
- Business days: Mon-Fri minus `holidays`. SQL `business_days_between` matches `lib/domain/bizdays.ts`.
- Money: Indian grouping and compact lakh form from `lib/domain/money.ts`.

## Events

- `saarang-2027` live simulated demo (editable).
- `campusfest-2025-work` working copy (editable).
- `campusfest-2025` original locked seed (chatbot test set).

const WINDOW_MS = 60_000;
const LIMIT = 20;
const hits = new Map<string, number[]>();

export function allowRequest(key: string, now = Date.now()): boolean {
  const recent = (hits.get(key) ?? []).filter((t) => now - t < WINDOW_MS);
  if (recent.length >= LIMIT) {
    hits.set(key, recent);
    return false;
  }
  recent.push(now);
  hits.set(key, recent);
  return true;
}

export type EventRow = {
  id: string;
  name: string;
  is_seed: boolean;
};

export async function listEvents(): Promise<EventRow[]> {
  const supabase = getServiceClient();
  const { data, error } = await supabase
    .from("events")
    .select("id, name, is_seed")
    .order("id");
  if (error) throw error;
  return (data ?? []) as EventRow[];
}

export async function getEvent(eventId: string): Promise<EventRow | null> {
  const supabase = getServiceClient();
  const { data, error } = await supabase
    .from("events")
    .select("id, name, is_seed")
    .eq("id", eventId)
    .maybeSingle();
  if (error) throw error;
  return data as EventRow | null;
}

const IST = "Asia/Kolkata";

export type Holiday = { on_date: string };

function toUtcDate(isoDate: string): Date {
  return new Date(`${isoDate}T00:00:00+05:30`);
}

function formatDateIST(date: Date): string {
  return new Intl.DateTimeFormat("en-CA", {
    timeZone: IST,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).format(date);
}

function isoDow(isoDate: string): number {
  const dow = toUtcDate(isoDate).getUTCDay();
  return dow === 0 ? 7 : dow;
}

export function isBusinessDay(isoDate: string, holidays: Holiday[]): boolean {
  const dow = isoDow(isoDate);
  if (dow > 5) return false;
  return !holidays.some((h) => h.on_date === isoDate);
}

export function businessDaysBetween(start: string, end: string, holidays: Holiday[]): number {
  if (end <= start) return 0;
  let count = 0;
  const cursor = toUtcDate(start);
  const last = toUtcDate(end);
  cursor.setUTCDate(cursor.getUTCDate() + 1);
  while (cursor.getTime() <= last.getTime()) {
    const iso = formatDateIST(cursor);
    if (isBusinessDay(iso, holidays)) count += 1;
    cursor.setUTCDate(cursor.getUTCDate() + 1);
  }
  return count;
}

export function addBusinessDays(start: string, days: number, holidays: Holiday[]): string {
  const cursor = toUtcDate(start);
  let remaining = days;
  while (remaining > 0) {
    cursor.setUTCDate(cursor.getUTCDate() + 1);
    const iso = formatDateIST(cursor);
    if (isBusinessDay(iso, holidays)) remaining -= 1;
  }
  return formatDateIST(cursor);
}

export function weekStartIST(isoDate: string): string {
  const cursor = toUtcDate(isoDate);
  const dow = isoDow(isoDate);
  cursor.setUTCDate(cursor.getUTCDate() - (dow - 1));
  return formatDateIST(cursor);
}

export function weekEndIST(isoDate: string): string {
  const start = toUtcDate(weekStartIST(isoDate));
  start.setUTCDate(start.getUTCDate() + 6);
  return formatDateIST(start);
}

export function businessDaysLeftInWeek(isoDate: string, holidays: Holiday[]): number {
  return businessDaysBetween(isoDate, weekEndIST(isoDate), holidays) + (isBusinessDay(isoDate, holidays) ? 1 : 0);
}

export function formatInr(amount: number): string {
  const rounded = Math.round(amount);
  const sign = rounded < 0 ? "-" : "";
  const abs = Math.abs(rounded).toString();
  const last3 = abs.slice(-3);
  const rest = abs.slice(0, -3);
  const grouped = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ",");
  return `₹${sign}${rest ? `${grouped},${last3}` : last3}`;
}

export function formatLakh(amount: number): string {
  return `₹${(amount / 100_000).toFixed(1)}L`;
}

===== FILE: lib\domain\rules-engine.ts =====
export type RuleOp = ">=" | "<=" | ">" | "<" | "=";
export type RuleClause = { m: string; op: RuleOp; v: number };
export type RuleJson = { all: RuleClause[] };

export function isRuleJson(value: unknown): value is RuleJson {
  if (!value || typeof value !== "object") return false;
  const all = (value as { all?: unknown }).all;
  if (!Array.isArray(all)) return false;
  return all.every(
    (clause) =>
      clause &&
      typeof clause === "object" &&
      typeof (clause as RuleClause).m === "string" &&
      typeof (clause as RuleClause).op === "string" &&
      typeof (clause as RuleClause).v === "number",
  );
}

===== FILE: lib\model\projection.ts =====
export type ProjectionInput = {
  signedToDate: number;
  openProspectInr: number;
  closeRate: number;
  inkindShare: number;
  prefestShare: number;
  avgNewDeal: number;
  warm: { leads: number; response: number; proposalRate: number; signRate: number };
  other: { leads: number; response: number; proposalRate: number; signRate: number };
};

export type ProjectionResult = {
  signed: number;
  cashBeforeFest: number;
};

export function project(input: ProjectionInput): ProjectionResult {
  const warmDeals =
    input.warm.leads * input.warm.response * input.warm.proposalRate * input.warm.signRate;
  const otherDeals =
    input.other.leads * input.other.response * input.other.proposalRate * input.other.signRate;
  const signed =
    input.signedToDate +
    input.openProspectInr * input.closeRate +
    (warmDeals + otherDeals) * input.avgNewDeal;
  const cashBeforeFest = signed * (1 - input.inkindShare) * input.prefestShare;
  return { signed, cashBeforeFest };
}

import { cookies } from "next/headers";

export const ROLE_COOKIE = "dd_role";
export const MEMBER_COOKIE = "dd_member";
export const EVENT_COOKIE = "dd_event";

export type DemoRole = "head" | "coordinator";

export const DEFAULT_EVENT_ID = "saarang-2027";

export async function getSession() {
  const store = await cookies();
  const role = (store.get(ROLE_COOKIE)?.value === "coordinator" ? "coordinator" : "head") as DemoRole;
  const memberId = store.get(MEMBER_COOKIE)?.value ?? null;
  const eventId = store.get(EVENT_COOKIE)?.value ?? DEFAULT_EVENT_ID;
  return { role, memberId, eventId };
}

import { GoogleGenAI } from "@google/genai";
import { DEFAULT_GEMINI_MODEL } from "../lib/ai/gemini";

config({ path: ".env.local" });

const TABLES = ["status_events", "weekly_targets", "fact_bank", "portfolio_gates"] as const;
const VIEWS = ["v_needs_update", "v_deal_desk", "v_initiative_funnel", "v_warmth_funnel"] as const;

function env(name: string): string | null {
  return process.env[name] ?? null;
}

async function relationExists(
  supabase: ReturnType<typeof createClient>,
  name: string,
): Promise<{ ok: boolean; detail: string }> {
  const { error } = await supabase.from(name).select("*", { head: true, count: "exact" }).limit(1);
  if (!error) return { ok: true, detail: "ok" };
  return { ok: false, detail: error.message };
}

async function main() {
  const url = env("NEXT_PUBLIC_SUPABASE_URL");
  const service = env("SUPABASE_SERVICE_ROLE_KEY");
  if (!url || !service) {
    console.log("FAIL: missing NEXT_PUBLIC_SUPABASE_URL or SUPABASE_SERVICE_ROLE_KEY (names only; values not printed)");
    process.exit(1);
  }

  const supabase = createClient(url, service, { auth: { persistSession: false } });
  const ping = await supabase.from("events").select("id", { head: true, count: "exact" }).limit(1);
  const connected = !ping.error || !/invalid api key|jwt|failed to fetch|enotfound/i.test(ping.error.message);

  console.log(connected ? "PASS connection" : `FAIL connection: ${ping.error?.message}`);

  let missing = 0;
  for (const table of TABLES) {
    const result = await relationExists(supabase, table);
    if (result.ok) console.log(`PASS table ${table}`);
    else {
      missing += 1;
      console.log(`FAIL table ${table}: ${result.detail}`);
    }
  }
  for (const view of VIEWS) {
    const result = await relationExists(supabase, view);
    if (result.ok) console.log(`PASS view ${view}`);
    else {
      missing += 1;
      console.log(`FAIL view ${view}: ${result.detail}`);
    }
  }

  const geminiKey = env("GEMINI_API_KEY");
  const model = env("GEMINI_MODEL") || DEFAULT_GEMINI_MODEL;
  if (!geminiKey) {
    console.log("FAIL gemini: GEMINI_API_KEY missing");
  } else {
    try {
      const ai = new GoogleGenAI({ apiKey: geminiKey });
      const response = await ai.models.generateContent({
        model,
        contents: "Reply with the single word OK",
      });
      const text = response.text ?? "";
      console.log(`PASS gemini model=${model} chars=${text.length}`);
    } catch (error: unknown) {
      const message = error instanceof Error ? error.message : "unknown error";
      console.log(`FAIL gemini model=${model}: ${message}`);
    }
  }

  if (!connected || missing > 0) {
    console.log("STOP: apply supabase/migrations/001_core.sql, 002_events_and_facts.sql, 003_views.sql in the Supabase SQL editor, then reply continue.");
    process.exit(1);
  }
  console.log("PASS schema 002/003 relations present");
}

main().catch((error: unknown) => {
  console.error(error instanceof Error ? error.message : error);
  process.exit(1);
});
