const WINDOW_MS = 60_000;
const LIMIT = 20;
const hits = new Map<string, number[]>();

export function allowRequest(key: string, now = Date.now()): boolean {
  const recent = (hits.get(key) ?? []).filter((t) => now - t < WINDOW_MS);
  if (recent.length >= LIMIT) {
    hits.set(key, recent);
    return false;
  }
  recent.push(now);
  hits.set(key, recent);
  return true;
}

export class HttpError extends Error {
  readonly status: number;
  constructor(status: number, message: string) {
    super(message);
    this.name = "HttpError";
    this.status = status;
  }
}

export class ForbiddenError extends HttpError {
  constructor(message = "Not allowed") {
    super(403, message);
    this.name = "ForbiddenError";
  }
}

export class NotFoundError extends HttpError {
  constructor(message = "Not found") {
    super(404, message);
    this.name = "NotFoundError";
  }
}

export class ValidationError extends HttpError {
  constructor(message = "Invalid input") {
    super(400, message);
    this.name = "ValidationError";
  }
}

export function errorStatus(error: unknown): number {
  return error instanceof HttpError ? error.status : 500;
}

export function errorMessage(error: unknown): string {
  if (error instanceof Error) return error.message;
  if (error && typeof error === "object" && "message" in error) {
    const message = (error as { message: unknown }).message;
    if (typeof message === "string") return message;
  }
  return "Unknown error";
}

export type CrmFilters = {

  channel?: string;
  tier?: string;
  needsUpdate?: boolean;

export type CrmRow = {
  id: string;
  name: string;
  crmStatus: string;
  channel: string | null;
  tier: string | null;
  amountInr: number;
  nextFollowUp: string | null;
  lastTouchAt: string | null;
  idleBdays: number;
  flag: "idle" | "follow_up_due" | null;
  poc: { name: string; jobTitle: string | null } | null;
};

export type CrmPage = { rows: CrmRow[]; total: number; page: number; pageSize: number };

async function firstContacts(brandIds: string[]): Promise<Map<string, { name: string; jobTitle: string | null }>> {
  const supabase = getServiceClient();
  const out = new Map<string, { name: string; jobTitle: string | null }>();
  for (let i = 0; i < brandIds.length; i += 100) {
    const { data, error } = await supabase
      .from("contacts")
      .select("brand_id, name, job_title, created_at")
      .in("brand_id", brandIds.slice(i, i + 100))
      .order("created_at");
    if (error) throw new Error(`contacts: ${error.message}`);
    for (const r of data ?? []) {
      const key = String(r.brand_id);
      if (!out.has(key)) out.set(key, { name: String(r.name), jobTitle: (r.job_title as string | null) ?? null });
    }
  }
  return out;
}

/** The coordinator's own companies (the head may pass an owner). Filtering, sorting and paging happen here. */
export async function listCompanies(ctx: Ctx, filters: CrmFilters = {}, ownerId?: string | null): Promise<CrmPage> {
  const owner = scopeOwner(ctx, ownerId);
  const eq: Record<string, string> = { event_id: ctx.eventId };
  if (owner) eq.owner_id = owner;
  const [brands, holidays, needs] = await Promise.all([
    fetchAll<Record<string, unknown>>(
      "brands",
      "id, name, crm_status, source_channel, company_tier, approx_amount_inr, next_follow_up, last_touch_at, created_at",
      eq,
    ),
    getHolidays(ctx.eventId),
    getNeedsUpdate(ctx, ownerId),
  ]);
  const live = brands.filter((b) => b.crm_status !== null);
  const flagOf = new Map(needs.map((n) => [n.brand_id, n.reason]));
  const contacts = await firstContacts(live.map((b) => String(b.id)));

  let rows: CrmRow[] = live.map((b) => {
    const id = String(b.id);
    const touch = (b.last_touch_at as string | null) ?? null;
    const from = toIstDate(touch ?? String(b.created_at));
    return {
      id,
      name: String(b.name),
      crmStatus: String(b.crm_status),
      channel: (b.source_channel as string | null) ?? null,
      tier: (b.company_tier as string | null) ?? null,
      amountInr: num(b.approx_amount_inr),
      nextFollowUp: (b.next_follow_up as string | null) ?? null,
      lastTouchAt: touch,
      idleBdays: businessDaysBetween(from, ctx.todayIst, holidays),
      flag: flagOf.get(id) ?? null,
      poc: contacts.get(id) ?? null,
    };
  });

  if (filters.status) rows = rows.filter((r) => r.crmStatus === filters.status);
  if (filters.channel) rows = rows.filter((r) => r.channel === filters.channel);
  if (filters.tier) rows = rows.filter((r) => r.tier === filters.tier);
  if (filters.needsUpdate) rows = rows.filter((r) => r.flag !== null);
  const q = (filters.q ?? "").trim().toLowerCase();
  if (q) rows = rows.filter((r) => r.name.toLowerCase().includes(q) || (r.poc?.name ?? "").toLowerCase().includes(q));

  const dir = filters.dir === "asc" ? 1 : -1;
  const key = filters.sort ?? "last_update";
  const cmp = (a: CrmRow, b: CrmRow): number => {
    switch (key) {
      case "company": return a.name.localeCompare(b.name);
      case "status": return a.crmStatus.localeCompare(b.crmStatus);
      case "amount": return a.amountInr - b.amountInr;
      case "follow_up": return (a.nextFollowUp ?? "9999").localeCompare(b.nextFollowUp ?? "9999");
      default: return a.idleBdays - b.idleBdays;
    }
  };
  rows.sort((a, b) => dir * cmp(a, b) || a.name.localeCompare(b.name));

  const pageSize = Math.max(1, Math.min(100, filters.pageSize ?? 25));
  const page = Math.max(1, filters.page ?? 1);
  return { rows: rows.slice((page - 1) * pageSize, page * pageSize), total: rows.length, page, pageSize };
}

export const OPEN_DEAL_STAGES = ["prospecting", "proposal_sent", "negotiating"] as const;
export type OpenDealStage = (typeof OPEN_DEAL_STAGES)[number];

/** Coordinators see their own open deals; closers see every open deal. */
export async function listOpenDeals(ctx: Ctx): Promise<DealDeskRow[]> {

async function loadDeal(ctx: Ctx, dealId: string): Promise<{ id: string; brandId: string; stage: string }> {
  const { data, error } = await getServiceClient().from("deals").select("id, event_id, owner_id, brand_id, stage").eq("id", dealId).maybeSingle();
  if (error) throw new Error(`deals: ${error.message}`);
  if (!data || data.event_id !== ctx.eventId) throw new NotFoundError("Deal not found.");
  if (ctx.role === "coordinator" && !ctx.isCloser && data.owner_id !== ctx.memberId) throw new ForbiddenError("This deal belongs to another coordinator.");
  if (!(OPEN_DEAL_STAGES as readonly string[]).includes(String(data.stage))) throw new ValidationError("This deal is already closed.");
  return { id: String(data.id), brandId: String(data.brand_id), stage: String(data.stage) };
}

/** Closing a deal goes through the company status (Converted or Rejected), so only open stages are editable here. */
export async function updateDeal(ctx: Ctx, dealId: string, patch: { stage?: OpenDealStage; nextFollowUp?: string | null }): Promise<void> {
  assertWritable(ctx);
  const deal = await loadDeal(ctx, dealId);
  const supabase = getServiceClient();
  const now = new Date().toISOString();
  const update: Record<string, unknown> = {};
  if (patch.stage && patch.stage !== deal.stage) {
    update.stage = patch.stage;
    update.stage_changed_at = now;
  }
  if (patch.nextFollowUp !== undefined) update.next_follow_up = patch.nextFollowUp;
  if (Object.keys(update).length === 0) return;
  const { error } = await supabase.from("deals").update(update).eq("id", deal.id);
  if (error) throw new Error(`deals: ${error.message}`);
  const brandPatch: Record<string, unknown> = {};
  if (update.stage) brandPatch.last_touch_at = now;
  if (patch.nextFollowUp !== undefined) brandPatch.next_follow_up = patch.nextFollowUp;
  const { error: bErr } = await supabase.from("brands").update(brandPatch).eq("id", deal.brandId);
  if (bErr) throw new Error(`brands: ${bErr.message}`);
}

/** One nudge: logs a follow-up activity and resets the idle clock. */

  if (bErr) throw new Error(`brands: ${bErr.message}`);
}

  follow_up: "Follow-up nudge",
  status_change: "Status change",
  outreach_sent: "Outreach sent",

const s = (v: unknown): string | null => (typeof v === "string" && v.length > 0 ? v : null);
const label = (code: string | null): string => (code && code in STATUS_LABELS ? STATUS_LABELS[code as StatusCode] : "New");

export async function getCompanyDetail(ctx: Ctx, brandId: string): Promise<CompanyDetail> {
  const supabase = getServiceClient();
  const { data: b, error } = await supabase
    .from("brands")
    .select(
      "id, event_id, name, industry, sector, owner_id, crm_status, source_channel, company_tier, approx_amount_inr, final_amount_inr, final_in_kind_inr, rejection_reason, next_follow_up, last_touch_at, notes, linkedin_url, call_attempts",
    )
    .eq("id", brandId)
    .maybeSingle();
  if (error) throw new Error(`brands: ${error.message}`);
  if (!b || b.event_id !== ctx.eventId || !b.crm_status) throw new NotFoundError("Company not found.");
  if (ctx.role === "coordinator" && b.owner_id !== ctx.memberId) throw new ForbiddenError("This company belongs to another coordinator.");

  const [contacts, events, acts, deals] = await Promise.all([
    supabase.from("contacts").select("id, name, job_title, email, phone, linkedin_url, created_at").eq("brand_id", brandId).order("created_at"),
    supabase.from("status_events").select("from_status, to_status, note, occurred_at").eq("brand_id", brandId).order("occurred_at", { ascending: false }).limit(40),
    supabase.from("activities").select("event_type, outcome, summary, occurred_at").eq("brand_id", brandId).order("occurred_at", { ascending: false }).limit(40),
    supabase
      .from("deals")
      .select("id, stage, proposed_value_inr, signed_value_inr, signed_cash_inr, signed_in_kind_inr, next_follow_up, created_at")
      .eq("brand_id", brandId)
      .order("created_at", { ascending: false })
      .limit(1),
  ]);
  for (const r of [contacts, events, acts, deals]) if (r.error) throw new Error(r.error.message);

  const deal = deals.data?.[0];
  let payments: PaymentRow[] = [];
  if (deal) {
    const p = await supabase.from("payments").select("id, amount_inr, due_date, phase, received_at").eq("deal_id", String(deal.id)).order("due_date");

        detail: s(a.summary),
      })),
  ].sort((x, y) => y.at.localeCompare(x.at));

  return {
    id: String(b.id),
    name: String(b.name),
    industry: s(b.industry),
    sector: s(b.sector),
    crmStatus: String(b.crm_status),
    channel: s(b.source_channel),
    tier: s(b.company_tier),
    approxAmountInr: num(b.approx_amount_inr),
    finalAmountInr: num(b.final_amount_inr),

    nextFollowUp: s(b.next_follow_up),
    lastTouchAt: s(b.last_touch_at),

/** Add or edit a point of contact on one of the coordinator's companies. */
export async function upsertContact(ctx: Ctx, brandId: string, input: PocInput): Promise<ContactRow> {
  assertWritable(ctx);
  const name = input.name.trim();
  if (!name) throw new ValidationError("A contact needs a name.");
  const email = input.email?.trim() || null;
  if (email && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) throw new ValidationError("That email address doesn't look right.");
  const supabase = getServiceClient();
  const { data: b, error: bErr } = await supabase.from("brands").select("id, event_id, owner_id").eq("id", brandId).maybeSingle();
  if (bErr) throw new Error(`brands: ${bErr.message}`);
  if (!b || b.event_id !== ctx.eventId) throw new NotFoundError("Company not found.");
  if (ctx.role === "coordinator" && b.owner_id !== ctx.memberId) throw new ForbiddenError("This company belongs to another coordinator.");
  const row = {
    brand_id: brandId,
    name,
    job_title: input.jobTitle?.trim() || null,
    email,
    phone: input.phone?.trim() || null,
    linkedin_url: input.linkedinUrl?.trim() || null,
  };
  const q = input.id
    ? supabase.from("contacts").update(row).eq("id", input.id).eq("brand_id", brandId)
    : supabase.from("contacts").insert(row);
  const { data, error } = await q.select("id, name, job_title, email, phone, linkedin_url").single();
  if (error) throw new Error(`contacts: ${error.message}`);
  return {
    id: String(data.id),
    name: String(data.name),
    jobTitle: s(data.job_title),
    email: s(data.email),
    phone: s(data.phone),
    linkedinUrl: s(data.linkedin_url),

async function createPaymentRows(ctx: Ctx, brandId: string): Promise<{ dealId: string | null; created: number }> {
  const supabase = getServiceClient();
  const { data: deals, error } = await supabase
    .from("deals")
    .select("id, signed_cash_inr, stage_changed_at")
    .eq("brand_id", brandId)
    .eq("stage", "partner_confirmed")
    .order("stage_changed_at", { ascending: false })
    .limit(1);
  if (error) throw new Error(`deals: ${error.message}`);
  const deal = deals?.[0];
  if (!deal) return { dealId: null, created: 0 };
  const dealId = String(deal.id);
  const { count, error: cErr } = await supabase.from("payments").select("id", { count: "exact", head: true }).eq("deal_id", dealId);
  if (cErr) throw new Error(`payments: ${cErr.message}`);
  if ((count ?? 0) > 0) return { dealId, created: 0 };
  const cash = num(deal.signed_cash_inr);
  if (cash <= 0) return { dealId, created: 0 };
  const plan = ctx.event.model_assumptions.plan;
  const raw = plan && typeof plan === "object" ? (plan as Record<string, unknown>).prefestShare : undefined;
  const share = typeof raw === "number" && raw > 0 && raw < 1 ? raw : 0.75;
  const pre = Math.round(cash * share);
  const start = ctx.event.starts_at ? toIstDate(ctx.event.starts_at) : ctx.todayIst;
  const end = ctx.event.ends_at ? toIstDate(ctx.event.ends_at) : start;
  const rows = [
    { deal_id: dealId, amount_inr: pre, due_date: addDays(start, -1), phase: "pre_fest", note: "Pre-fest instalment (assumed 75%)" },
    { deal_id: dealId, amount_inr: cash - pre, due_date: addDays(end, 15), phase: "post_fest", note: "Post-fest instalment (assumed 25%)" },
  ];
  const { error: iErr } = await supabase.from("payments").insert(rows);
  if (iErr) throw new Error(`payments: ${iErr.message}`);
  return { dealId, created: rows.length };
}

export async function changeCompanyStatus(ctx: Ctx, input: StatusChangeInput): Promise<{ dealId: string | null; paymentsCreated: number }> {
  assertWritable(ctx);
  const brand = await loadOwnedBrand(ctx, input.brandId);
  const problem = validateTransition(brand.crm_status, input);
  if (problem) throw new ValidationError(problem);

  const now = new Date().toISOString();
  const patch: Record<string, unknown> = { crm_status: input.to, last_touch_at: now };
  if (OUTREACH_STATUSES.includes(input.to) && !brand.first_outreach_at) patch.first_outreach_at = now;
  if (RESPONSE_STATUSES.includes(input.to) && !brand.first_response_at) patch.first_response_at = now;
  if (input.note) patch.notes = input.note;

  if (input.to === "prospective") {
    patch.approx_amount_inr = input.approxAmountInr;
    const holidays = await getHolidays(ctx.eventId);
    patch.next_follow_up = input.nextFollowUp ?? addBusinessDays(ctx.todayIst, 3, holidays);
  } else if (input.nextFollowUp !== undefined) {
    patch.next_follow_up = input.nextFollowUp;
  }
  if (input.to === "converted") {
    patch.final_amount_inr = input.finalAmountInr;
    patch.final_in_kind_inr = input.inKindInr ?? 0;
    patch.next_follow_up = null;
  }
  if (input.to === "rejected") {
    patch.rejection_reason = (input.rejectionReason ?? "").trim();
    patch.next_follow_up = null;
  }

  const supabase = getServiceClient();
  const { error } = await supabase.from("brands").update(patch).eq("id", brand.id);
  if (error) throw friendly(error.message);
  await logStatusActivity(ctx, brand, input.note ?? null, input.to);

  if (input.to === "converted") {
    const res = await createPaymentRows(ctx, brand.id);
    return { dealId: res.dealId, paymentsCreated: res.created };
  }
  return { dealId: null, paymentsCreated: 0 };
}

function toLines(value: unknown): string[] {

  if (typeof value === "string") return value.trim() ? [value.trim()] : [];
  if (Array.isArray(value)) return value.flatMap((v) => (typeof v === "object" && v ? [JSON.stringify(v)] : [String(v)]));
  if (typeof value === "object") {

const pct = (x: number): string => `${Math.round(x * 1000) / 10}%`;

export function buildModelRows(model: PlanModel): ModelRow[] {
  const { settings: s, state: st, projection: p } = model;
  const seg = (leads: number, r: { response: number; proposalRate: number }) => ({
    conv: leads * r.response,
    prop: leads * r.response * r.proposalRate,
  });
  const bw = seg(st.queueWarm, s.base.warm);
  const bo = seg(st.queueOther, s.base.other);
  const sw = seg(s.stretchWarmLeads, s.stretch.warm);
  const so = seg(s.stretchOtherLeads, s.stretch.other);
  const eff = (r: ProjectionResult): number =>
    r.signed > 0 && s.prefestShare > 0 ? 1 - r.cashBeforeFest / (r.signed * s.prefestShare) : 0;
  const row = (key: string, label: string, unit: ModelRow["unit"], floor: number, base: number, stretch: number): ModelRow => ({
    key,
    label,
    unit,
    floor,
    base,
    stretch,
  });
  return [
    row("new_warm", "New outreach: warm leads", "count", 0, st.queueWarm, s.stretchWarmLeads),
    row("new_other", "New outreach: other leads", "count", 0, st.queueOther, s.stretchOtherLeads),
    row("conversations", "Conversations (responses)", "count", 0, bw.conv + bo.conv, sw.conv + so.conv),
    row("proposals", "Proposals (prospective or better)", "count", 0, bw.prop + bo.prop, sw.prop + so.prop),
    row("new_deals", "New signed deals", "deals", 0, p.base.newDeals, p.stretch.newDeals),
    row("open_closes", "Open-prospect closes", "deals", 0, p.base.openDeals, p.stretch.openDeals),
    row("total_deals", "Total signed deals", "deals", st.signedDealsToDate, p.base.totalDeals, p.stretch.totalDeals),
    row("total_signed", "Total signed amount", "inr", st.signedToDate, p.base.signed, p.stretch.signed),
    row("inkind", "In-kind share (effective)", "pct", st.observedInkindShare, eff(p.base), eff(p.stretch)),
    row("cash", "Cash before fest", "inr", p.floor, p.base.cashBeforeFest, p.stretch.cashBeforeFest),
  ];
}

export function workedExample(model: PlanModel): string[] {
  const { settings: s, state: st, projection: p } = model;
  const line = (name: string, r: ProjectionResult, closeRate: number, inkind: number): string[] => [
    `${name} signed = ${formatInr(st.signedToDate)} signed + ${formatInr(st.openProspectInr)} x ${pct(closeRate)} open-prospect close + ${r.newDeals.toFixed(2)} new deals x ${formatInr(s.avgNewDeal)} = ${formatInr(r.signed)}`,
    `${name} cash before fest = ${formatInr(r.signed)} x (1 - ${pct(inkind)} in-kind) x ${pct(s.prefestShare)} pre-fest = ${formatInr(r.cashBeforeFest)}`,
  ];
  const out = [
    ...line("Base", p.base, s.base.closeRate, s.base.inkindShare),
    ...line("Stretch", p.stretch, s.stretch.closeRate, s.stretch.inkindShare),
    `Floor cash = ${formatInr(st.signedToDate)} x (1 - ${pct(st.observedInkindShare)} observed in-kind) x ${pct(s.prefestShare)} pre-fest = ${formatInr(p.floor)}`,
  ];
  if (s.inkindScope === "new_only") {
    out.push("In-kind policy is new_only: existing signed keeps its observed in-kind share, so the cash lines above are approximate.");
  }
  return out;
}

export function observedRows(model: PlanModel, warmth: WarmthFunnelRow[]): ObservedRow[] {
  const out: ObservedRow[] = [];
  for (const seg of ["warm", "other"] as const) {
    const w = warmth.find((r) => r.warmth === seg);
    const o = Number(w?.outreached ?? 0);
    const r = Number(w?.responded ?? 0);
    const p = Number(w?.prospective_or_better ?? 0);
    const c = Number(w?.converted ?? 0);
    const b = model.settings.base[seg];
    const t = model.settings.stretch[seg];
    const mk = (metric: string, base: number, stretch: number, top: number, den: number): ObservedRow => ({
      segment: seg,
      metric,
      base,
      stretch,
      observed: den > 0 ? top / den : null,
      n: den,
      outreached: o,
      enough: o >= MIN_OBSERVED_N,
    });

      mk("Response rate (responded / outreached)", b.response, t.response, r, o),
      mk("Proposal rate (prospective or better / responded)", b.proposalRate, t.proposalRate, p, r),
      mk("Sign rate (converted / prospective or better)", b.signRate, t.signRate, c, p),
    );
  }
  return out;
}

function summaryLines(value: unknown): string[] {
  if (!value || typeof value !== "object" || Array.isArray(value)) return [];
  return Object.entries(value as Row).flatMap(([k, v]) => {
    const key = k.replace(/_/g, " ");
    if (typeof v === "number" || typeof v === "string" || typeof v === "boolean") return [`${key}: ${String(v)}`];
    if (Array.isArray(v)) return [`${key}: ${v.length}`];
    return [];
  });
}

export type SlabPackage = { name: string; priceInr: number; headline: string | null };
export type SlabItem = { id: string; name: string; category: string; priceMin: number; priceMax: number };
export type SlabSuggestion = { pkg: SlabPackage | null; addons: SlabItem[]; spentInr: number; leftoverInr: number };

/** Largest package within budget, then unsold catalogue add-ons (at minimum price) with what is left. */
export function suggestSlab(budget: number, packages: SlabPackage[], unsold: SlabItem[]): SlabSuggestion {
  if (!(budget > 0)) return { pkg: null, addons: [], spentInr: 0, leftoverInr: 0 };
  const pkg = [...packages].filter((p) => p.priceInr <= budget).sort((a, b) => b.priceInr - a.priceInr)[0] ?? null;
  let left = budget - (pkg?.priceInr ?? 0);
  const addons: SlabItem[] = [];
  for (const it of [...unsold].sort((a, b) => b.priceMin - a.priceMin)) {
    if (addons.length >= 6) break;
    if (it.priceMin > 0 && it.priceMin <= left) {
      addons.push(it);
      left -= it.priceMin;
    }
  }
  return { pkg, addons, spentInr: budget - left, leftoverInr: left };
}

export async function getInitiativeStatuses(ctx: Ctx): Promise<InitiativeStatus[]> {
  requireHead(ctx);
  const [campaigns, funnel, desk, signed, brands, drafts, events] = await Promise.all([
    fetchAll<Record<string, unknown>>("campaigns", "id, source_id, name, scale_rule, change_rule, stop_rule", {
      event_id: ctx.eventId,
      kind: "initiative",
    }),
    getInitiativeFunnel(ctx),
    getDealDesk(ctx),
    getSignedDeals(ctx),
    getBrandRows(ctx),
    fetchAll<Record<string, unknown>>("email_drafts", "brand_id, personalised, status", { event_id: ctx.eventId }),
    fetchAll<Record<string, unknown>>("status_events", "brand_id, to_status", { event_id: ctx.eventId }),
  ]);

  const today = dateNum(ctx.todayIst);
  const deskRows = desk.map((r) => ({
    company: r.company,
    stage: String(r.stage),
    proposed_inr: num(r.proposed_inr),
    idle_bdays: num(r.idle_bdays),
    nudges: num(r.nudges),
  }));
  const draftRows = drafts.map((r) => ({ brand_id: String(r.brand_id), personalised: Boolean(r.personalised), status: String(r.status) }));
  const eventRows = events.map((r) => ({ brand_id: String(r.brand_id), to_status: String(r.to_status) }));

  const out: InitiativeStatus[] = [];
  for (const c of campaigns) {
    const campaignId = String(c.id);
    const code = String(c.source_id);
    let set: MetricSet = { metrics: { today }, counts: {} };
    let experiment: Experiment | null = null;

    if (code === "bet-1") {
      set = dealDeskMetrics(deskRows, today);
    } else if (code === "bet-2") {
      set = cashTermsMetrics(signed, PLAN_START, today);
    } else {
      set = laneMetrics(funnel.find((f) => f.initiative_id === campaignId), today);
      if (code === "bet-5") {
        const ids = new Set(brands.filter((b) => b.initiative_id === campaignId).map((b) => b.id));
        experiment = experimentArms(ids, draftRows, eventRows);
        set = {
          metrics: { ...set.metrics, personalised_vs_template_ratio: experiment.ratio },
          counts: {
            ...set.counts,
            personalised_vs_template_ratio:
              experiment.ratio === undefined
                ? `personalised ${experiment.personalised.n} sent vs template ${experiment.template.n} sent`
                : `personalised ${experiment.personalised.reached} of ${experiment.personalised.n} vs template ${experiment.template.reached} of ${experiment.template.n} reached proposal`,
          },
        };
      }
    }

    const rules: InitiativeRules = { scale: asRule(c.scale_rule), change: asRule(c.change_rule), stop: asRule(c.stop_rule) };
    const evaluation = evaluateInitiative(code as CodeArg, rules, set.metrics, set.counts);
    out.push({
      campaignId,
      code,
      name: String(c.name),
      badge: evaluation.badge,
      reason: evaluation.reason,
      evaluation,
      metrics: set.metrics,
      experiment,
    });
  }
  return out.sort((a, b) => a.code.localeCompare(b.code));
}

/** Reads every row of a table or view (1000 per page), filtered by equality. */
export async function fetchAll<T>(table: string, select: string, eq: Record<string, string> = {}): Promise<T[]> {
  const supabase = getServiceClient();
  const out: T[] = [];
  for (let from = 0; ; from += 1000) {
    let query = supabase.from(table).select(select).range(from, from + 999);
    for (const [column, value] of Object.entries(eq)) query = query.eq(column, value);
    const { data, error } = await query;
    if (error) throw new Error(`${table}: ${error.message}`);
    const rows = (data ?? []) as unknown as T[];
    out.push(...rows);
    if (rows.length < 1000) break;
  }
  return out;
}

/** Everything the coordinator weekly strip needs, from one place. */
export async function getWeeklyStrip(ctx: Ctx, memberId?: string | null): Promise<WeeklyStrip> {
  const owner = scopeOwner(ctx, memberId ?? ctx.memberId);
  if (!owner) throw new ValidationError("A coordinator must be selected.");
  const weekStart = weekStartIST(ctx.todayIst);
  const [holidays, tgt, progress] = await Promise.all([
    getHolidays(ctx.eventId),
    getWeeklyTarget(ctx, owner, weekStart),
    getWeeklyProgress(ctx, owner),
  ]);
  const row = progress.find((r) => r.week_start === weekStart);
  const outreached = row?.outreached ?? 0;
  const responded = row?.responded ?? 0;
  const left = businessDaysLeftInWeek(ctx.todayIst, holidays);
  const remaining = Math.max(0, tgt.target - outreached);
  return {
    weekStart,
    weekEnd: weekEndIST(weekStart),
    target: tgt.target,
    label: tgt.label,
    source: tgt.source,
    outreached,
    responded,
    responseRate: outreached > 0 ? responded / outreached : null,
    businessDaysLeft: left,
    remaining,
    requiredPerDay: left > 0 ? remaining / left : null,
  };

function formatMs(ms: number): string {
  const d = new Date(ms);
  const y = d.getUTCFullYear();
  const m = String(d.getUTCMonth() + 1).padStart(2, "0");
  const day = String(d.getUTCDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

function holidaySet(holidays: Holiday[]): Set<string> {
  return new Set(holidays.map((h) => h.day));
}

export function toIstDate(value: string | Date): string {
  return istFormat.format(typeof value === "string" ? new Date(value) : value);
}

export function addDays(iso: string, days: number): string {
  return formatMs(parseIso(iso) + days * DAY_MS);
}

export function daysBetween(start: string, end: string): number {
  return Math.round((parseIso(end) - parseIso(start)) / DAY_MS);
}

export function isoDow(iso: string): number {
  const dow = new Date(parseIso(iso)).getUTCDay();
  return dow === 0 ? 7 : dow;
}

export function isBusinessDay(iso: string, holidays: Holiday[] = []): boolean {
  return isoDow(iso) <= 5 && !holidaySet(holidays).has(iso);
}

/** Business days AFTER start, up to and including end (matches SQL business_days_between). */
export function businessDaysBetween(start: string, end: string, holidays: Holiday[] = []): number {
  if (end <= start) return 0;
  const hs = holidaySet(holidays);
  const last = parseIso(end);
  let count = 0;
  for (let ms = parseIso(start) + DAY_MS; ms <= last; ms += DAY_MS) {
    const dow = new Date(ms).getUTCDay();
    if (dow !== 0 && dow !== 6 && !hs.has(formatMs(ms))) count += 1;
  }
  return count;
}

export function addBusinessDays(start: string, days: number, holidays: Holiday[] = []): string {
  const hs = holidaySet(holidays);
  let ms = parseIso(start);
  let remaining = Math.max(0, Math.floor(days));
  while (remaining > 0) {
    ms += DAY_MS;
    const dow = new Date(ms).getUTCDay();
    if (dow !== 0 && dow !== 6 && !hs.has(formatMs(ms))) remaining -= 1;
  }
  return formatMs(ms);
}

export function weekStartIST(iso: string): string {
  return addDays(iso, -(isoDow(iso) - 1));
}

export function weekEndIST(iso: string): string {
  return addDays(weekStartIST(iso), 6);
}

/** Business days left in the Mon-Sun week, counting today if it is a business day. */
export function businessDaysLeftInWeek(iso: string, holidays: Holiday[] = []): number {
  return businessDaysBetween(iso, weekEndIST(iso), holidays) + (isBusinessDay(iso, holidays) ? 1 : 0);
}

  if (TERMINAL_STATUSES.includes(from)) return false;

    case "send_mail":
      return from === "to_contact";
    case "slabs_sent":
    case "didnt_pick":
      return from === "to_contact" || from === "send_mail";
    case "ghosting":
      return from === "contacted" || from === "prospective";
    case "converted":
      return from === "prospective" || from === "contacted" || from === "ghosting";
    default:
      return true;
  }
}

export function validateTransition(from: StatusCode, input: StatusChangeInput): string | null {
  const to = input.to;
  if (TERMINAL_STATUSES.includes(from)) return `${STATUS_LABELS[from]} is final, so this company can't be moved.`;
  if (!isLegalTransition(from, to)) {
    if (to === "didnt_pick") return "Didn't Pick is only valid from To Contact or Send Mail.";
    if (to === "ghosting") return "Ghosting is only valid from Contacted or Prospective.";
    if (to === "converted") return "Converted needs the company to be Prospective, Contacted or Ghosting first.";

    if (!(final > 0)) return "Converted needs the final amount.";
    if (inKind < 0 || inKind > final) return "In-kind must be between zero and the final amount.";
  }

function show(names: string[], metrics: Metrics, counts: Record<string, string>): string {
  const parts = names
    .filter((n) => n !== "today")
    .map((n) => counts[n] ?? (metrics[n] === undefined ? `${n} unknown` : `${n} ${Number(metrics[n].toFixed(3))}`));
  return parts.join(", ");
}

function minOutreach(rules: InitiativeRules): number | null {
  const vals: number[] = [];
  for (const r of [rules.scale, rules.change, rules.stop]) {
    for (const c of r?.all ?? []) if (c.m === "outreached" && (c.op === ">=" || c.op === ">")) vals.push(c.v);
  }
  return vals.length ? Math.min(...vals) : null;
}

export function evaluateInitiative(
  code: string,
  rules: InitiativeRules,
  metrics: Metrics,
  counts: Record<string, string> = {},
): BadgeResult {
  const stop = evalRule(rules.stop, metrics);
  if (stop.met) return { badge: "STOP", reason: `Stop rule met: ${show(ruleMetricNames(rules.stop), metrics, counts) || "date rule reached"}.` };
  const change = evalRule(rules.change, metrics);
  if (change.met) return { badge: "CHANGE", reason: `Change rule met: ${show(ruleMetricNames(rules.change), metrics, counts)}.` };
  const scale = evalRule(rules.scale, metrics);
  if (scale.met) return { badge: "SCALE", reason: `Scale rule met: ${show(ruleMetricNames(rules.scale), metrics, counts)}.` };

  const today = metrics.today ?? 0;
  if (code === "bet-1" && today > dateNum(BET1_SCALE_BY) && (metrics.negotiating_count ?? 0) < 5) {
    return { badge: "BEHIND", reason: `Behind: ${metrics.negotiating_count ?? 0} negotiating, 5 needed by ${BET1_SCALE_BY}.` };
  }
  const need = minOutreach(rules);
  const out = metrics.outreached;
  if (need !== null && out !== undefined && out < need) {
    return { badge: "TOO EARLY", reason: `Too early: ${out} outreached, rules need ${need}.` };
  }
  const unknown = [stop, change, scale].flatMap((r) => r.clauses.filter((c) => c.met === null).map((c) => c.clause.m));
  const tail = unknown.length ? ` Insufficient data for: ${[...new Set(unknown)].join(", ")}.` : "";
  const all = [...new Set([...ruleMetricNames(rules.scale), ...ruleMetricNames(rules.change), ...ruleMetricNames(rules.stop)])];
  return { badge: "HOLD", reason: `No rule met: ${show(all, metrics, counts) || "no data yet"}.${tail}` };
}

function groupIndian(amount: number): string {
  const digits = Math.abs(Math.round(amount)).toString();
  if (digits.length <= 3) return digits;
  const last3 = digits.slice(-3);
  const rest = digits.slice(0, -3).replace(/\B(?=(\d{2})+(?!\d))/g, ",");

}

export function formatInr(amount: number | null | undefined): string {
  if (amount === null || amount === undefined || Number.isNaN(amount)) return "-";
  const sign = Math.round(amount) < 0 ? "-" : "";
  return `${sign}${RUPEE}${groupIndian(amount)}`;
}

export function formatLakh(amount: number | null | undefined): string {
  if (amount === null || amount === undefined || Number.isNaN(amount)) return "-";
  const sign = amount < 0 ? "-" : "";
  return `${sign}${RUPEE}${(Math.abs(amount) / 100_000).toFixed(1)}L`;
}

export function parseInr(input: string): number | null {
  const cleaned = input.replace(/[\u20B9,\s]/g, "").toLowerCase();
  if (!cleaned) return null;
  const lakh = cleaned.endsWith("l");
  const value = Number(lakh ? cleaned.slice(0, -1) : cleaned);
  if (!Number.isFinite(value)) return null;
  return Math.round(lakh ? value * 100_000 : value);
}

export type RuleOp = ">=" | "<=" | ">" | "<" | "=";
export type RuleClause = { m: string; op: RuleOp; v: number };
export type RuleJson = { all: RuleClause[] };
export type Metrics = Record<string, number | undefined>;

export const RULE_METRICS = [
  "outreached",
  "response_rate",
  "proposal_rate",
  "prospective_or_better",
  "signed_deals",
  "negotiating_count",
  "top5_max_idle_bdays",
  "deals_idle_gt10_after_3_nudges",
  "inkind_share_new",
  "personalised_vs_template_ratio",
  "today",
] as const;

const OPS: RuleOp[] = [">=", "<=", ">", "<", "="];

export function isRuleJson(value: unknown): value is RuleJson {
  if (!value || typeof value !== "object") return false;
  const all = (value as { all?: unknown }).all;
  if (!Array.isArray(all)) return false;
  return all.every(
    (clause) =>
      clause &&
      typeof clause === "object" &&
      typeof (clause as RuleClause).m === "string" &&
      typeof (clause as RuleClause).op === "string" &&
      typeof (clause as RuleClause).v === "number",
  );
}

export function validateRuleJson(value: unknown): string[] {
  if (value === null) return [];
  if (!isRuleJson(value)) return ["Rule must look like {\"all\": [{\"m\": metric, \"op\": op, \"v\": number}]}"];
  const errors: string[] = [];
  if (value.all.length === 0) errors.push("A rule needs at least one clause");
  value.all.forEach((c, i) => {
    if (!(RULE_METRICS as readonly string[]).includes(c.m)) errors.push(`Clause ${i + 1}: unknown metric "${c.m}"`);
    if (!OPS.includes(c.op)) errors.push(`Clause ${i + 1}: unknown operator "${c.op}"`);
    if (!Number.isFinite(c.v)) errors.push(`Clause ${i + 1}: value must be a finite number`);
  });
  return errors;
}

export function compare(op: RuleOp, a: number, b: number): boolean {
  switch (op) {
    case ">=": return a >= b;
    case "<=": return a <= b;
    case ">": return a > b;
    case "<": return a < b;
    default: return Math.abs(a - b) < 1e-9;
  }
}

export type ClauseResult = { clause: RuleClause; value: number | undefined; met: boolean | null };
export type RuleResult = { met: boolean | null; clauses: ClauseResult[] };

export function evalRule(rule: RuleJson | null | undefined, metrics: Metrics): RuleResult {
  if (!rule) return { met: false, clauses: [] };
  const clauses = rule.all.map((clause): ClauseResult => {
    const value = metrics[clause.m];
    const ok = value === undefined || Number.isNaN(value) ? null : compare(clause.op, value, clause.v);
    return { clause, value: ok === null ? undefined : value, met: ok };
  });
  if (clauses.some((c) => c.met === false)) return { met: false, clauses };
  if (clauses.some((c) => c.met === null)) return { met: null, clauses };
  return { met: true, clauses };
}

export function ruleMetricNames(rule: RuleJson | null | undefined): string[] {
  return rule ? [...new Set(rule.all.map((c) => c.m))] : [];
}

export function dateNum(iso: string): number {
  return Number(iso.replace(/-/g, ""));
}

export function clauseText(c: RuleClause): string {
  return `${c.m} ${c.op} ${c.v}`;
}

function merge<T>(defaults: T, raw: unknown): T {
  if (isObj(defaults)) {
    const src = isObj(raw) ? raw : {};
    const out: Record<string, unknown> = {};
    for (const key of Object.keys(defaults)) out[key] = merge(defaults[key], src[key]);
    return out as T;

  }
  return (typeof raw === typeof defaults ? raw : defaults) as T;
}

export function attributeStretch(state: PlanState, settings: ModelSettings = DEFAULT_SETTINGS): Attribution {
  const input = scenarioInput(state, "stretch", settings);
  const res = project(input);
  const obs = state.observedInkindShare;
  const p = settings.prefestShare;
  const st = settings.stretch;
  const avg = settings.avgNewDeal;
  const keep = 1 - obs;
  const perLead = (r: { response: number; proposalRate: number; signRate: number }) => r.response * r.proposalRate * r.signRate;

  const bet1 = state.openProspectInr * st.closeRate * keep * p;
  const inkindBase = settings.inkindScope === "all" ? res.signed : res.signed - state.signedToDate;
  const bet2 = inkindBase * (obs - st.inkindShare) * p;
  const bet3Leads = Math.max(0, settings.stretchWarmLeads - settings.artistLeads);
  const bet3 = bet3Leads * perLead(st.warm) * avg * keep * p;
  const bet4 = settings.artistLeads * perLead(st.warm) * avg * keep * p;
  const bet5 = segmentDeals({ leads: settings.stretchOtherLeads, ...st.other }) * avg * keep * p;

    { bet: "bet-2", label: "Cash-first terms", cashOverFloor: Math.round(bet2) },

export function experimentArms(
  brandIds: Set<string>,
  drafts: { brand_id: string; personalised: boolean; status: string }[],
  events: { brand_id: string; to_status: string }[],
): Experiment {

    const hit = ids.filter((id) => reached.has(id)).length;
    return { n: ids.length, reached: hit, rate: ids.length ? hit / ids.length : null };
  };
  const personalised = stats(true);
  const template = stats(false);
  const ratio = personalised.rate !== null && template.rate !== null && template.rate > 0 ? personalised.rate / template.rate : undefined;
  return { personalised, template, ratio, directionalOnly: personalised.n < 100 || template.n < 100 };
}

export type ProjectionInput = {
  signedToDate: number;
  openProspectInr: number;
  closeRate: number;
  inkindShare: number;
  prefestShare: number;
  avgNewDeal: number;
  warm: SegmentInput;
  other: SegmentInput;
  observedInkindShare?: number;
  inkindScope?: InkindScope;
  signedDealsToDate?: number;
  openProspectCount?: number;

export type ProjectionResult = {
  signed: number;
  cashBeforeFest: number;
  floorCash: number;
  newSigned: number;
  openClosed: number;
  newDeals: number;
  openDeals: number;
  totalDeals: number;
};

const fix = (x: number): number => Math.round(Math.round(x * 1e6) / 1e6);

export const segmentDeals = (s: SegmentInput): number => s.leads * s.response * s.proposalRate * s.signRate;

export function floorCash(input: Pick<ProjectionInput, "signedToDate" | "prefestShare" | "inkindShare" | "observedInkindShare">): number {
  const obs = input.observedInkindShare ?? input.inkindShare;
  return fix(input.signedToDate * (1 - obs) * input.prefestShare);
}

export function project(input: ProjectionInput): ProjectionResult {
  const obs = input.observedInkindShare ?? input.inkindShare;
  const scope = input.inkindScope ?? "all";
  const newDeals = segmentDeals(input.warm) + segmentDeals(input.other);
  const newSigned = newDeals * input.avgNewDeal;
  const openClosed = input.openProspectInr * input.closeRate;
  const signed = input.signedToDate + openClosed + newSigned;
  const keep = 1 - input.inkindShare;
  const cash =
    scope === "all"
      ? signed * keep * input.prefestShare
      : (input.signedToDate * (1 - obs) + (openClosed + newSigned) * keep) * input.prefestShare;
  const openDeals = (input.openProspectCount ?? 0) * input.closeRate;
  return {
    signed: fix(signed),
    cashBeforeFest: fix(cash),
    floorCash: floorCash(input),
    newSigned: fix(newSigned),
    openClosed: fix(openClosed),
    newDeals: Math.round(newDeals * 1e6) / 1e6,
    openDeals,
    totalDeals: (input.signedDealsToDate ?? 0) + openDeals + newDeals,
  };
}

export type DemoRole = "head" | "coordinator";

export const DEFAULT_EVENT_ID = "saarang-2027";

export async function getSession() {
  const store = await cookies();
  const role = (store.get(ROLE_COOKIE)?.value === "coordinator" ? "coordinator" : "head") as DemoRole;
  const memberId = store.get(MEMBER_COOKIE)?.value ?? null;
  const eventId = store.get(EVENT_COOKIE)?.value ?? DEFAULT_EVENT_ID;
  return { role, memberId, eventId };
}
