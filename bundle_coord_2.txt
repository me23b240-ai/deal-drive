const EVENTS = [
  { id: "saarang-2027", label: "Saarang 2027" },
  { id: "campusfest-2025-work", label: "CampusFest work" },
  { id: "campusfest-2025", label: "CampusFest original" },
] as const;

export function EventSwitch(props: { eventId: string }) {
  return (
    <div className="flex flex-wrap items-center gap-2 text-sm">
      {EVENTS.map((event) => (
        <span
          key={event.id}
          className={
            event.id === props.eventId
              ? "rounded-full border border-accent px-3 py-1 text-accent"
              : "rounded-full border border-border px-3 py-1 text-muted"
          }
        >
          {event.label}
        </span>
      ))}
      {props.eventId === "campusfest-2025" ? (
        <span className="rounded border border-approx/40 bg-approx/10 px-2 py-1 text-xs text-approx">
          Locked original
        </span>
      ) : null}
    </div>
  );
}

"use client";

import type { DemoRole } from "@/lib/session";

export function RoleToggle(props: { role: DemoRole }) {
  const isHead = props.role === "head";
  return (
    <div
      role="radiogroup"
      aria-label="Role"
      className="relative flex h-11 min-h-11 w-[220px] items-center rounded-full border-2 border-accent/40 bg-card p-1"
    >
      <span
        aria-hidden
        className="absolute top-1 bottom-1 w-[calc(50%-4px)] rounded-full bg-accent/20 transition-transform duration-200"
        style={{ transform: isHead ? "translateX(0)" : "translateX(100%)" }}
      />
      <a
        href="/head"
        role="radio"
        aria-checked={isHead}
        className={`relative z-10 flex h-full flex-1 items-center justify-center text-sm ${isHead ? "text-foreground" : "text-muted"}`}
      >
        Head
      </a>
      <a
        href="/coordinator"
        role="radio"
        aria-checked={!isHead}
        className={`relative z-10 flex h-full flex-1 items-center justify-center text-sm ${!isHead ? "text-foreground" : "text-muted"}`}
      >
        Coordinator
      </a>
    </div>
  );
}
